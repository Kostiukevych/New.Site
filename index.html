🎨 3D-модель персонажа

Основний персонаж проєкту створений у додатку VRoid Mobile.

  • Метадані моделі: автор — Dmytro, версія — VRM 0.x (формат класичного VRM0).
  • Список кісток (55 Humanoid Bones): у моделі присутні всі 55 стандартних кісток (включно з upperChest та повним набором фаланг пальців). Іменування внутрішніх вузлів виконане за схемою J_Bip_... (наприклад, hips ➔ J_Bip_C_Hips, head ➔ J_Bip_C_Head, leftHand ➔ J_Bip_L_Hand тощо).
  • Список виразів обличчя (15 груп): Neutral; голосні A, I, U, E, O; очі Blink, Blink_L, Blink_R; емоції Angry, Fun, Joy, Sorrow, Surprised, Extra (кастомний слот).

⚠️ Важлива примітка щодо сумісності VRM 0.x та VRM 1.0:

Оскільки модель створена у форматі VRM0, а конвертер json_to_vrma.py генерує анімації за специфікацією VRM 1.0, під час створення JSON для мімики варто враховувати мапінг назв пресетів: A ➔ aa, I ➔ ih, U ➔ ou, E ➔ ee, O ➔ oh; Joy ➔ happy, Fun ➔ relaxed, Sorrow ➔ sad, Angry ➔ angry, Surprised ➔ surprised.

⚙️ Власний конвертер анімацій: json_to_vrma.py

Щоб не залежати від складних програм, було створено спеціальний Python-скрипт. Він приймає простий JSON-файл із ключовими кадрами та конвертує його у повноцінний бінарний файл .vrma.

json_to_vrma.py 
#!/usr/bin/env python3
"""
json_to_vrma.py  (v2 — кости + мимика)

Конвертирует JSON с ключевыми кадрами в бинарный .vrma (GLB +
расширение VRMC_vrm_animation). Работает с любой VRM-моделью,
включая экспортированные из VRoid Studio / VRoid Mobile —
т.к. использует СТАНДАРТНЫЕ имена костей VRM Humanoid, а не
внутренние имена нодов модели (J_Bip_C_Hips и т.п.).

Установка (в т.ч. в Termux):
    pkg install python          # если ещё не стоит
    pip install pygltflib

Запуск:
    python json_to_vrma.py input.json output.vrma
    python json_to_vrma.py --list-bones      # показать список стандартных имён костей
    python json_to_vrma.py --list-expr       # показать список стандартных имён мимики
"""

import sys
import json
import struct
from pygltflib import (
    GLTF2, Asset, Scene, Node, Animation, AnimationSampler, AnimationChannel,
    AnimationChannelTarget, Buffer, BufferView, Accessor,
    FLOAT, SCALAR, VEC3, VEC4
)

VRM_SPEC_VERSION = "1.0"

# Стандартные имена костей VRM 1.0 Humanoid.
# Обязательна только "hips", остальные — по необходимости.
STANDARD_BONES = [
    "hips", "spine", "chest", "upperChest", "neck", "head",
    "leftEye", "rightEye", "jaw",
    "leftUpperLeg", "leftLowerLeg", "leftFoot", "leftToes",
    "rightUpperLeg", "rightLowerLeg", "rightFoot", "rightToes",
    "leftShoulder", "leftUpperArm", "leftLowerArm", "leftHand",
    "rightShoulder", "rightUpperArm", "rightLowerArm", "rightHand",
    "leftThumbMetacarpal", "leftThumbProximal", "leftThumbDistal",
    "leftIndexProximal", "leftIndexIntermediate", "leftIndexDistal",
    "leftMiddleProximal", "leftMiddleIntermediate", "leftMiddleDistal",
    "leftRingProximal", "leftRingIntermediate", "leftRingDistal",
    "leftLittleProximal", "leftLittleIntermediate", "leftLittleDistal",
    "rightThumbMetacarpal", "rightThumbProximal", "rightThumbDistal",
    "rightIndexProximal", "rightIndexIntermediate", "rightIndexDistal",
    "rightMiddleProximal", "rightMiddleIntermediate", "rightMiddleDistal",
    "rightRingProximal", "rightRingIntermediate", "rightRingDistal",
    "rightLittleProximal", "rightLittleIntermediate", "rightLittleDistal",
]

# Стандартные пресеты мимики VRM 1.0 (preset expressions)
STANDARD_EXPRESSIONS = [
    "happy", "angry", "sad", "relaxed", "surprised", "neutral",
    "blink", "blinkLeft", "blinkRight",
    "aa", "ih", "ou", "ee", "oh",          # положения рта (визема)
    "lookUp", "lookDown", "lookLeft", "lookRight",
]


def load_config(path):
    with open(path, "r", encoding="utf-8") as f:
        return json.load(f)


def pack_floats(values):
    flat = []
    for v in values:
        if isinstance(v, (list, tuple)):
            flat.extend(v)
        else:
            flat.append(v)
    return struct.pack("<%df" % len(flat), *flat)


def add_accessor(gltf, buffer_blob, data, component_type, accessor_type, min_max=None):
    raw = pack_floats(data)
    byte_offset = len(buffer_blob)
    pad = (-len(raw)) % 4
    raw = raw + b"\x00" * pad
    buffer_blob += raw

    bv = BufferView(buffer=0, byteOffset=byte_offset, byteLength=len(raw) - pad)
    gltf.bufferViews.append(bv)
    bv_index = len(gltf.bufferViews) - 1

    acc = Accessor(bufferView=bv_index, componentType=component_type,
                    count=len(data), type=accessor_type)
    if min_max:
        acc.min, acc.max = min_max
    gltf.accessors.append(acc)
    return len(gltf.accessors) - 1, buffer_blob


def add_track(gltf, anim, buffer_blob, node_index, path, keyframes):
    """path: 'rotation' | 'translation' | 'scale'"""
    times = [kf["time"] for kf in keyframes]
    values = [kf["value"] for kf in keyframes]

    time_acc, buffer_blob = add_accessor(
        gltf, buffer_blob, times, FLOAT, SCALAR,
        min_max=([min(times)], [max(times)]),
    )
    vec_type = VEC4 if path == "rotation" else VEC3
    val_acc, buffer_blob = add_accessor(gltf, buffer_blob, values, FLOAT, vec_type)

    sampler = AnimationSampler(input=time_acc, output=val_acc, interpolation="LINEAR")
    anim.samplers.append(sampler)
    sampler_index = len(anim.samplers) - 1

    channel = AnimationChannel(
        sampler=sampler_index,
        target=AnimationChannelTarget(node=node_index, path=path),
    )
    anim.channels.append(channel)
    return buffer_blob


def build_vrma(config, output_path):
    bones_cfg = config.get("bones", {})
    expr_cfg = config.get("expressions", {})
    if not bones_cfg and not expr_cfg:
        raise ValueError("В JSON нет ни 'bones', ни 'expressions'.")

    gltf = GLTF2()
    gltf.asset = Asset(version="2.0", generator="json_to_vrma.py")
    gltf.scenes = [Scene(nodes=[])]
    gltf.scene = 0
    gltf.nodes = []
    gltf.animations = [Animation(channels=[], samplers=[])]
    anim = gltf.animations[0]
    buffer_blob = b""

    human_bones_ext = {}

    # --- Кости тела ---
    for bone_name, tracks in bones_cfg.items():
        if bone_name not in STANDARD_BONES:
            print(f"[Внимание] '{bone_name}' не входит в стандартные имена VRM Humanoid "
                  f"(проверь --list-bones). Всё равно добавляю.")
        node_index = len(gltf.nodes)
        gltf.nodes.append(Node(name=bone_name))
        gltf.scenes[0].nodes.append(node_index)
        human_bones_ext[bone_name] = {"node": node_index}

        for prop in ("rotation", "translation", "scale"):
            if prop in tracks:
                buffer_blob = add_track(gltf, anim, buffer_blob, node_index, prop, tracks[prop])

    # --- Мимика (expressions) ---
    preset_ext = {}
    custom_ext = {}
    for group_name in ("preset", "custom"):
        for expr_name, keyframes in expr_cfg.get(group_name, {}).items():
            group_name_eff = group_name
            if group_name == "preset" and expr_name not in STANDARD_EXPRESSIONS:
                print(f"[Внимание] '{expr_name}' не входит в стандартные пресеты мимики "
                      f"(проверь --list-expr). Кладу как custom.")
                group_name_eff = "custom"

            node_index = len(gltf.nodes)
            gltf.nodes.append(Node(name=f"expr_{expr_name}"))
            gltf.scenes[0].nodes.append(node_index)

            # Вес мимики (0..1) кодируется через translation.x — так делает
            # официальный конвертер UniVRM, т.к. в glTF нет анимации веса
            # морф-таргетов для "виртуального" нода без меша.
            weight_kf = [{"time": kf["time"], "value": [kf["value"], 0, 0]} for kf in keyframes]
            buffer_blob = add_track(gltf, anim, buffer_blob, node_index, "translation", weight_kf)

            if group_name_eff == "preset":
                preset_ext[expr_name] = {"node": node_index}
            else:
                custom_ext[expr_name] = {"node": node_index}

    gltf.buffers = [Buffer(byteLength=len(buffer_blob))]
    gltf.set_binary_blob(buffer_blob)

    ext = {
        "specVersion": VRM_SPEC_VERSION,
        "humanoid": {"humanBones": human_bones_ext},
    }
    if preset_ext or custom_ext:
        ext["expressions"] = {}
        if preset_ext:
            ext["expressions"]["preset"] = preset_ext
        if custom_ext:
            ext["expressions"]["custom"] = custom_ext

    gltf.extensionsUsed = ["VRMC_vrm_animation"]
    gltf.extensions = {"VRMC_vrm_animation": ext}

    gltf.save_binary(output_path)
    print(f"Готово: {output_path}")
    print(f"Костей: {len(bones_cfg)}, мимики: {len(preset_ext) + len(custom_ext)}")


def main():
    if "--list-bones" in sys.argv:
        print("\n".join(STANDARD_BONES))
        return
    if "--list-expr" in sys.argv:
        print("\n".join(STANDARD_EXPRESSIONS))
        return
    if len(sys.argv) != 3:
        print("Использование:")
        print("  python json_to_vrma.py input.json output.vrma")
        print("  python json_to_vrma.py --list-bones")
        print("  python json_to_vrma.py --list-expr")
        sys.exit(1)

    config = load_config(sys.argv[1])
    build_vrma(config, sys.argv[2])


if __name__ == "__main__":
    main()

📱 Як створити та оживити власну 3D-анімацію

Увесь процес створення рухів відбувається прямо на смартфоні у терміналі Termux. Для цього не потрібен комп'ютер чи складні 3D-редактори!

📋 Крок 1. Запит до ШІ (Генерація JSON)

Щоб отримати код руху, потрібно попросити штучний інтелект (Gemini, Claude або DeepSeek) написати анімацію. Дайте нейромережі: шаблон-приклад example_wave_face.json; список доступних кісток (python json_to_vrma.py --list-bones); список виразів обличчя (python json_to_vrma.py --list-expr); опис руху (наприклад: «Персонаж маше правою рукою 2 секунди та посміхається»). Важливо: нейромережа має повернути чистий JSON-код із кватерніонами повороту кісток.

📂 Крок 2. Збереження JSON-файлу у робочу папку

Отриманий JSON-файл потрібно покласти в ту ж саму папку, де знаходиться сам Python-скрипт json_to_vrma.py. Це можна зробити через редактор nano у Termux (nano dance.json → вставити код → зберегти CTRL+O, Enter → вийти CTRL+X), або скопіювати файл із папки завантажень телефону в робочу папку зі скриптом.

Termux Commands 
pkg install python
pip install pygltflib
nano dance.json
python json_to_vrma.py dance.json dance.vrma

⚙️ Крок 3. Компіляція у бінарний .vrma

Переходимо в робочу папку (де вже лежать і json_to_vrma.py, і новий .json файл) та запускаємо конвертер. Готово! Скрипт згенерує файл dance.vrma, який одразу можна підключити до 3D-моделі — і персонаж відтворить створений рух!
